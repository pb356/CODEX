cmake_minimum_required(VERSION 3.20)
project(DesktopBoxes VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
 add_compile_options(/utf-8 /W4 /permissive-)
else()
 add_compile_options(-Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(core-tests tests/core_tests.cpp)
add_test(NAME core-model COMMAND core-tests)
if(WIN32)
 enable_language(RC)
 add_compile_definitions(UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
 add_executable(DesktopBoxes WIN32 src/main.cpp src/app.rc)
 add_executable(WallpaperPreview WIN32 src/wallpaper_preview.cpp src/app.rc)
 add_custom_command(TARGET WallpaperPreview POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
  "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:WallpaperPreview>/assets")
 target_link_libraries(WallpaperPreview PRIVATE gdiplus d2d1 windowscodecs shell32 ole32 uuid user32 gdi32 wtsapi32 comctl32)
 add_executable(DesktopIntegrationProbe tools/desktop_probe.cpp)
 target_link_libraries(DesktopIntegrationProbe PRIVATE shell32 ole32 oleaut32 uuid advapi32)
 target_include_directories(DesktopBoxes PRIVATE src)
 target_link_libraries(DesktopBoxes PRIVATE gdiplus shell32 ole32 uuid comdlg32 wtsapi32 dwmapi gdi32 user32 advapi32 comctl32 uxtheme oleacc oleaut32)
 add_executable(windows-tests tests/windows_tests.cpp)
 target_link_libraries(windows-tests PRIVATE shell32 ole32 uuid)
 add_test(NAME windows-persistence COMMAND windows-tests)
 add_executable(ui-smoke tests/ui_smoke.cpp src/app.rc)
 target_link_libraries(ui-smoke PRIVATE gdiplus shell32 ole32 uuid comdlg32 wtsapi32 dwmapi gdi32 user32 advapi32 comctl32 uxtheme oleacc oleaut32)
 add_test(NAME win32-ui-smoke COMMAND ui-smoke)
 add_executable(features-tests tests/features_tests.cpp src/app.rc)
 target_link_libraries(features-tests PRIVATE gdiplus shell32 ole32 uuid comdlg32 wtsapi32 dwmapi gdi32 user32 advapi32 comctl32 uxtheme oleacc oleaut32)
 add_test(NAME win32-features COMMAND features-tests)
 add_executable(wallpaper-tests tests/wallpaper_tests.cpp src/app.rc)
 target_link_libraries(wallpaper-tests PRIVATE gdiplus d2d1 windowscodecs shell32 ole32 uuid user32 gdi32 wtsapi32 comctl32)
 add_dependencies(wallpaper-tests WallpaperPreview)
 add_test(NAME win32-wallpaper-preview COMMAND wallpaper-tests)
 if(MINGW)
  target_link_options(ui-smoke PRIVATE -municode -static -static-libgcc -static-libstdc++)
  target_link_options(features-tests PRIVATE -municode -static -static-libgcc -static-libstdc++)
  target_link_options(wallpaper-tests PRIVATE -municode -static -static-libgcc -static-libstdc++)
  target_link_options(DesktopBoxes PRIVATE -municode -static -static-libgcc -static-libstdc++)
  target_link_options(WallpaperPreview PRIVATE -municode -static -static-libgcc -static-libstdc++)
  target_link_options(DesktopIntegrationProbe PRIVATE -municode -static -static-libgcc -static-libstdc++)
  target_link_options(core-tests PRIVATE -static -static-libgcc -static-libstdc++)
  target_link_options(windows-tests PRIVATE -municode -static -static-libgcc -static-libstdc++)
 endif()
endif()
